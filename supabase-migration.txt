-- 5.1 Función para generar folio de venta
CREATE OR REPLACE FUNCTION public.generar_folio_venta()
RETURNS varchar AS $$
DECLARE
    v_anio smallint;
    v_mes smallint;
    v_folio integer;
    v_folio_completo varchar(20);
BEGIN
    v_anio := EXTRACT(YEAR FROM now())::smallint;
    v_mes := EXTRACT(MONTH FROM now())::smallint;
    
    -- Insertar o actualizar el contador
    INSERT INTO public.folio_venta (anio, mes, ultimo_folio)
    VALUES (v_anio, v_mes, 1)
    ON CONFLICT (anio, mes) 
    DO UPDATE SET ultimo_folio = public.folio_venta.ultimo_folio + 1
    RETURNING ultimo_folio INTO v_folio;
    
    -- Formatear: AAAA-MM-NNNNNN
    v_folio_completo := v_anio::text || '-' || LPAD(v_mes::text, 2, '0') || '-' || LPAD(v_folio::text, 6, '0');
    
    RETURN v_folio_completo;
END;
$$ LANGUAGE plpgsql;

-- 5.2 Función para verificar si cliente tiene deuda pendiente
CREATE OR REPLACE FUNCTION public.cliente_tiene_deuda(p_clienteid integer)
RETURNS boolean AS $$
DECLARE
    v_tiene_deuda boolean;
BEGIN
    SELECT EXISTS (
        SELECT 1 FROM public.cobros 
        WHERE clienteid = p_clienteid 
        AND estado = 1  -- pendiente
    ) INTO v_tiene_deuda;
    
    RETURN v_tiene_deuda;
END;
$$ LANGUAGE plpgsql;

-- 5.3 Función para obtener total deuda de cliente
CREATE OR REPLACE FUNCTION public.obtener_deuda_cliente(p_clienteid integer)
RETURNS numeric AS $$
DECLARE
    v_total_deuda numeric;
BEGIN
    SELECT COALESCE(SUM(total), 0) 
    INTO v_total_deuda
    FROM public.cobros 
    WHERE clienteid = p_clienteid 
    AND estado = 1;  -- pendiente
    
    RETURN v_total_deuda;
END;
$$ LANGUAGE plpgsql;

-- 5.4 Trigger para actualizar fechaactualizacion automáticamente
CREATE OR REPLACE FUNCTION public.trigger_set_fechaactualizacion()
RETURNS TRIGGER AS $$
BEGIN
    NEW.fechaactualizacion = now();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 8.1 Vista de ventas con detalle completo
CREATE OR REPLACE VIEW public.v_ventas_completas AS
SELECT 
    v.idventa,
    v.folio,
    v.fecha,
    v.total,
    v.tipoventa,
    v.metodopago,
    v.montopagado,
    v.cambio,
    v.estado,
    v.observacion,
    v.fecharegistro,
    c.idcliente,
    c.cliente as nombre_cliente,
    c.ci_nit,
    c.telefono as telefono_cliente,
    u.idusuario,
    u.primernombre || ' ' || u.apellidopaterno as vendedor
FROM public.venta v
LEFT JOIN public.cliente c ON v.clienteid = c.idcliente
LEFT JOIN public.usuario u ON v.usuarioid = u.idusuario;

-- 8.2 Vista de cobros con información completa
CREATE OR REPLACE VIEW public.v_cobros_completos AS
SELECT 
    co.idcobro,
    co.nombrecobro,
    co.telefono,
    co.total,
    co.estado,
    co.origen,
    co.observacion,
    co.fechacreacion,
    co.fechapago,
    co.imagen,
    c.idcliente,
    c.cliente as nombre_cliente,
    c.ci_nit,
    v.idventa,
    v.folio as folio_venta,
    u.idusuario,
    u.primernombre || ' ' || u.apellidopaterno as vendedor
FROM public.cobros co
LEFT JOIN public.cliente c ON co.clienteid = c.idcliente
LEFT JOIN public.venta v ON co.ventaid = v.idventa
LEFT JOIN public.usuario u ON co.usuarioid = u.idusuario;

-- 8.3 Vista de productos con sus unidades y precios
CREATE OR REPLACE VIEW public.v_productos_completos AS
SELECT 
    p.idproducto,
    p.nombreproducto,
    p.descripcion,
    p.imagen,
    p.estado,
    p.fechacreacion,
    cat.idcategoria,
    cat.nombrecategoria,
    json_agg(
        json_build_object(
            'idproductounidad', pu.idproductounidad,
            'unidadid', um.idunidad,
            'nombre_unidad', um.nombre,
            'abreviatura', um.abreviatura,
            'es_peso', um.es_peso,
            'precio', pu.precio
        )
    ) FILTER (WHERE pu.idproductounidad IS NOT NULL) as unidades
FROM public.producto p
LEFT JOIN public.categoria cat ON p.categoriaid = cat.idcategoria
LEFT JOIN public.producto_unidad pu ON p.idproducto = pu.productoid AND pu.estado = 1
LEFT JOIN public.unidad_medida um ON pu.unidadid = um.idunidad
WHERE p.estado = 1
GROUP BY p.idproducto, cat.idcategoria;

-- 8.4 Vista resumen dashboard
CREATE OR REPLACE VIEW public.v_resumen_dashboard AS
SELECT
    (SELECT COUNT(*) FROM public.venta WHERE estado = 1 AND DATE(fecha) = CURRENT_DATE) as ventas_hoy,
    (SELECT COALESCE(SUM(total), 0) FROM public.venta WHERE estado = 1 AND DATE(fecha) = CURRENT_DATE) as total_ventas_hoy,
    (SELECT COUNT(*) FROM public.venta WHERE estado = 1) as total_ventas,
    (SELECT COALESCE(SUM(total), 0) FROM public.venta WHERE estado = 1) as total_ingresos,
    (SELECT COUNT(*) FROM public.cobros WHERE estado = 1) as cobros_pendientes,
    (SELECT COALESCE(SUM(total), 0) FROM public.cobros WHERE estado = 1) as total_cobros_pendientes,
    (SELECT COUNT(*) FROM public.pedido WHERE estado = 1) as pedidos_pendientes,
    (SELECT COUNT(*) FROM public.producto WHERE estado = 1) as total_productos,
    (SELECT COUNT(*) FROM public.cliente WHERE estado = 1 AND es_generico = false) as total_clientes;
